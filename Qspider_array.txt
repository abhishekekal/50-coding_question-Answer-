48.Sorting elements of an array by frequency

int[] a = { 20, 30, 20, 40, 60, 40, 80, 90, 80};
        int[] f = new int[a.length];
        int v = -1;

        // Step 1: Calculate frequency of each element
        for (int i = 0; i < a.length; i++) {
            int c = 1;
            for (int j = i + 1; j < a.length; j++) {
                if (a[i] == a[j]) {
                    c++;
                    f[j] = v; // Mark this index as visited
                }
            }
            if (f[i] != v) {
                f[i] = c;
            }
        }

        // Step 2: Store elements and their frequencies in a list
        List<int[]> flist = new ArrayList<>();
        for (int i = 0; i < f.length; i++) {
            if (f[i] != v) {
                flist.add(new int[]{a[i], f[i]});
            }
        }

        // Step 3: Sort the list by frequency in descending order
        Collections.sort(flist, new Comparator<int[]>() {
            public int compare(int[] o1, int[] o2) {
                return Integer.compare(o2[1], o1[1]); // Sort by frequency descending
            }
        });

        // Step 4: Print the elements based on their frequency
        for (int[] freq : flist) {
            for (int i = 0; i < freq[1]; i++) {
                System.out.print(freq[0] + " ");
            }
        }
		
49. Longest palindrome in an array

import java.util.*;
class Main {
    
    public static boolean ispalindrome(int n){
        int rem,num,rev=0;
        num=n;
        while(n!=0){
            rem=n%10;
            rev=rev*10+rem;
            n=n/10;
        }
        return num==rev;
    }
    
    public static int largestpalindrome(int a[],int n){
       Arrays.sort(a);
       
       for(int i=n-1;i>=0;i--){
           if(ispalindrome(a[i])){
               return a[i];
           }
       }
       return  -1;
       
        
    }
    public static void main(String[] args) {
             int[] a = {123, 454, 789, 121, 99899, 34543, 11111};
             int s=a.length;
             int largestp=largestpalindrome(a,s);
             
             if(largestp ==-1){
                 System.out.println("no");
             }else{
                 System.out.println("yes -: "+ largestp);
             }
             
             
             
    }
}

Implement Odd-Even sort /Brick sort.

import java.util.Scanner;

public class SimpleOddEvenSort {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);

        // Input: Number of elements
        int n = scanner.nextInt();

        // Input: Array elements
        int[] arr = new int[n];
        for (int i = 0; i < n; i++) {
            arr[i] = scanner.nextInt();
        }

        // Perform Odd-Even Sort
        boolean sorted = false;
        while (!sorted) {
            sorted = true;

            // Odd phase
            for (int i = 1; i < n - 1; i += 2) {
                if (arr[i] > arr[i + 1]) {
                    // Swap
                    int temp = arr[i];
                    arr[i] = arr[i + 1];
                    arr[i + 1] = temp;
                    sorted = false;
                }
            }

            // Even phase
            for (int i = 0; i < n - 1; i += 2) {
                if (arr[i] > arr[i + 1]) {
                    // Swap
                    int temp = arr[i];
                    arr[i] = arr[i + 1];
                    arr[i + 1] = temp;
                    sorted = false;
                }
            }
        }

        // Output: Sorted array
        for (int num : arr) {
            System.out.print(num + " ");
        }

        scanner.close();
    }
}

	   public class RotateMatrix {
    public static void rotate90Degree(int[][] matrix) {
        int n = matrix.length;

        // Step 1: Transpose the matrix
        for (int i = 0; i < n; i++) {
            for (int j = i; j < n; j++) {
                int temp = matrix[i][j];
                matrix[i][j] = matrix[j][i];
                matrix[j][i] = temp;
            }
        }

        // Step 2: Reverse each row
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n / 2; j++) {
                int temp = matrix[i][j];
                matrix[i][j] = matrix[i][n - 1 - j];
                matrix[i][n - 1 - j] = temp;
            }
        }
    }

    public static void printMatrix(int[][] matrix) {
        for (int[] row : matrix) {
            for (int num : row) {
                System.out.print(num + " ");
            }
            System.out.println();
        }
    }

    public static void main(String[] args) {
        int[][] matrix = {
            {1, 2, 3},
            {4, 5, 6},
            {7, 8, 9}
        };

        System.out.println("Original Array:");
        printMatrix(matrix);

        rotate90Degree(matrix);

        System.out.println("\n90 Degree Rotated Array:");
        printMatrix(matrix);
    }
}

        arr[start] = arr[end];

        arr[end] = temp;

        reverse(arr, start + 1, end - 1);

    }

// Approach 1: Sort + Manual Split

import java.util.*;



public class Approach1 {

    public static void main(String[] args) {

        int[] arr = {5, 2, 9, 1, 7, 6, 3};

        int[] result = rearrangeWithManualSplit(arr);

        System.out.println("Approach 1 (Manual Split): " + Arrays.toString(result));

    }



    public static int[] rearrangeWithManualSplit(int[] input) {

        int[] arr = Arrays.copyOf(input, input.length);

        Arrays.sort(arr);

        int[] result = new int[arr.length];

        int mid = arr.length / 2;

        for (int i = 0; i < mid; i++) {

            result[i] = arr[i];

        }

        int index = mid;

        for (int i = arr.length - 1; i >= mid; i--) {

            result[index++] = arr[i];

        }

        return result;

    }

}

    }



    public static List<Integer> rearrangeWithCollections(int[] input) {

        Integer[] boxed = Arrays.stream(input).boxed().toArray(Integer[]::new);

        Arrays.sort(boxed);

        List<Integer> result = new ArrayList<>();

        int mid = boxed.length / 2;

            result.add(boxed[i]);

        int[] sorted = Arrays.stream(input).sorted().toArray();

        int n = arr.length;

// ✅ Method 3: In-place Reversal Algorithm (Right Rotation)

public class RotateRightInPlace {

    public static void rotateRightInPlace(int[] arr, int k) {

        int n = arr.length;

        k %= n;



        reverse(arr, 0, n - 1);      // Reverse the whole array

        reverse(arr, 0, k - 1);      // Reverse the first k elements

        reverse(arr, k, n - 1);      // Reverse the remaining elements

    }



    private static void reverse(int[] arr, int start, int end) {

        while (start < end) {

            int temp = arr[start];

            arr[start++] = arr[end];

            arr[end--] = temp;

        }

    }



    public static void main(String[] args) {

        int[] arr = {1, 2, 3, 4, 5, 6};

        int k = 2;

        rotateRightInPlace(arr, k);

        System.out.println("Array after right rotation (in-place): " + Arrays.toString(arr));

    }

}
